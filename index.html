// =========================================================
// GOOGLE APPS SCRIPT CODE (Code.gs)
// วางโค้ดนี้ใน Google Sheets > Extensions > Apps Script
// =========================================================

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000);
  
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    if (sheet.getLastRow() === 0) {
      sheet.appendRow([
        "Timestamp", "Order ID", "Customer Name", "Phone", 
        "Address", "Subdistrict/District/Province", "Zipcode", 
        "Payment Method", "Total Amount (THB)", "Items Purchased", 
        "Status", "Courier", "Tracking No"
      ]);
      sheet.getRange(1, 1, 1, 13).setFontWeight("bold").setBackground("#14362B").setFontColor("#ffffff");
    }
    
    var data = JSON.parse(e.postData.contents);
    var itemsStr = data.items.map(function(item) {
      return item.sku + " (" + item.name + ") x" + item.qty;
    }).join("\n");
    
    sheet.appendRow([
      new Date(), data.orderId, data.customerName, "'" + data.phone,
      data.address, data.districtInfo || "-", "'" + data.zipcode, 
      data.paymentMethod, data.totalAmount, itemsStr, 
      data.status || "PENDING_PICK", data.courier || "FLASH", data.trackingNo || ""
    ]);
    
    return ContentService.createTextOutput(JSON.stringify({"result": "success", "orderId": data.orderId}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({"result": "error", "error": err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}

function doGet(e) {
  // Returns order list as JSON for warehouse sync
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = sheet.getDataRange().getValues();
  var headers = data[0];
  var orders = [];
  
  for (var i = 1; i < data.length; i++) {
    var row = data[i];
    orders.push({
      date: row[0], orderId: row[1], customerName: row[2], phone: row[3],
      address: row[4], districtInfo: row[5], zipcode: row[6],
      paymentMethod: row[7], totalAmount: row[8], itemsRaw: row[9],
      status: row[10], courier: row[11], trackingNo: row[12]
    });
  }
  
  return ContentService.createTextOutput(JSON.stringify(orders))
    .setMimeType(ContentService.MimeType.JSON);
}
